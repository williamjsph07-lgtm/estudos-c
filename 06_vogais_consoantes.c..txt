#include <stdio.h>
#include <ctype.h>

int main() {
    char letra;
    int vogais = 0;
    int consoantes = 0;

    printf("Digite letras (digite 0 para parar):\n");

    while (1) {
        scanf(" %c", &letra);

        if (letra == '0') {
            break;
        }

        letra = tolower(letra);

        if (letra >= 'a' && letra <= 'z') {
            if (letra == 'a' || letra == 'e' || letra == 'i' ||
                letra == 'o' || letra == 'u') {
                vogais++;
            } else {
                consoantes++;
            }
        }
    }

    printf("\nVogais: %d\n", vogais);
    printf("Consoantes: %d\n", consoantes);

    return 0;
}
